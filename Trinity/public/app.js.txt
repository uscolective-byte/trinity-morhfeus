async function api(path,body){const r=await fetch('/api/ops'+path,{method:body===undefined?'GET':'POST',headers:body===undefined?{}:{'Content-Type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const data=await r.json();if(!r.ok){if(r.status===401)lock();throw new Error(data.error||'Požiadavka zlyhala.');}return data;}

async function watch(id,progress,text,box){const tick=async()=>{try{const j=await api('/jobs/'+id);progress.textContent=`${j.context_summary?'Pamäť: '+j.context_summary.memories.length+' záznamov · ':''}${labels[j.status]||j.status} · ${j.steps.filter(s=>s.status==='completed').length}/${j.team.length} krokov · ${j.steps.filter(s=>s.status==='running').map(s=>agents.find(a=>a.id===s.agent_id)?.name||s.agent_id).join(', ')}`;if(j.status==='completed'){text.textContent=j.result;box.append(processSummary(j));const a=el('a','Stiahnuť celý výstup ↓');a.href='/api/ops/jobs/'+id+'/artifact';box.append(a);await start();}else if(['failed','cancelled'].includes(j.status)){text.textContent=j.error||'Úloha bola zrušená.';text.classList.add('failed');}else poll=setTimeout(tick,3500);}catch(e){notice(e.message);}};clearTimeout(poll);await tick();}
