function formatMarkdown(text){
  if(!text)return '';
  const d=document.createElement('div');
  d.textContent=text;
  let s=d.innerHTML;
  s=s.replace(/```([a-zA-Z0-9_-]*)\n([\s\S]*?)```/g,'<pre class="code-block"><code class="lang-$1">$2</code></pre>');
  s=s.replace(/`([^`\n]+)`/g,'<code class="inline-code">$1</code>');
  s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
  s=s.replace(/(^|[^*])\*([^*\n]+)\*/g,'$1<em>$2</em>');
  s=s.replace(/^[ \t]*[\*\-][ \t]+(.+)$/gm,'<li>$1</li>');
  s=s.replace(/(<li>[\s\S]*?<\/li>)/g,'<ul class="md-list">$1</ul>');
  s=s.replace(/<\/ul>\s*<ul class="md-list">/g,'');
  s=s.replace(/\n\n+/g,'<br><br>');
  return s;
}
function renderReply(target,text){
  target.innerHTML=formatMarkdown(text);
  const match=text.match(/\/api\/assistant\/media\/images\/[0-9a-f-]{36}/i);
  if(match){
    const image=document.createElement('img');
    image.className='generated-media';
    image.src=match[0];
    image.alt='Obrázok vytvorený Trinity';
    image.loading='lazy';
    target.append(image);
  }
  const fileMatches=Array.from(text.matchAll(/\/api\/assistant\/files\/([0-9a-f-]{36})\/([a-zA-Z0-9._\-]+)/gi));
  if(fileMatches.length){
    for(const fm of fileMatches){
      const url=fm[0];
      const fname=fm[2];
      const ext=fname.split('.').pop()?.toLowerCase();
      let icon='📄';
      if(ext==='pdf')icon='📕';
      else if(ext==='py')icon='🐍';
      else if(ext==='json')icon='📦';
      else if(['js','ts','mjs'].includes(ext))icon='⚡';
      else if(['jpg','png','webp','gif','svg'].includes(ext))icon='🖼️';
      else if(ext==='csv')icon='📊';
      const fileCard=document.createElement('div');
      fileCard.className='generated-file-card';
      fileCard.innerHTML=`<span class="file-icon">${icon}</span><div class="file-info"><b>${fname}</b><small>Vytvorený súbor v R2 archíve</small></div><a href="${url}?download=1" download="${fname}" class="file-download-btn">⬇ Stiahnuť</a><a href="${url}" target="_blank" rel="noopener" class="file-view-btn">Otvoriť ↗</a>`;
      target.append(fileCard);
    }
  }
}

function renderSelfImprovement(data){
  const status=$('self-improvement-deploy-status'),list=$('self-improvement-list');
  status.textContent=data.capabilities?.deployment||'Nasadenie sa riadi GitHub Actions.';
  list.replaceChildren();
  if(!data.items?.length){list.append(node('p','Zatiaľ nie sú evidované žiadne návrhy.','skeleton'));return;}
  for(const item of data.items){
    const card=node('article',undefined,'api-key-item'),info=node('div');
    info.append(node('b',item.title),node('small',`${item.kind} · ${item.status}`),node('p',item.description));
    card.append(info);
    if(item.plugin_id)card.append(node('small',`Modul: ${item.plugin_id}`));
    if(item.status==='pending'&&data.can_approve===true){
      const approve=node('button','Schváliť návrh','primary');approve.type='button';
      approve.onclick=safe(async()=>{await api('/self-improvement/'+item.id+'/approve',{});await loadSettings();notice('Návrh bol schválený a zapísaný do auditu.');});
      card.append(approve);
    }
    if(item.status==='approved'&&!item.result_json){
      const apply=node('button',item.kind==='plugin_install'?'Nainštalovať modul':'Vykonať návrh','primary');apply.type='button';
      apply.onclick=safe(async()=>{await api('/self-improvement/'+item.id+'/apply',{});await loadSettings();notice('Návrh bol spracovaný. Výsledok a prípadné zostávajúce kroky sú pri návrhu.');});
      card.append(apply);
    }
    if(item.result_json){info.append(node('small',item.result_json.status||'Spracované'));if(item.result_json.url){const link=node('a','Otvoriť výsledok');link.href=item.result_json.url;link.target='_blank';link.rel='noopener noreferrer';info.append(link);}if(item.result_json.error)info.append(node('p',item.result_json.error));}
    list.append(card);
  }
}

function renderSettingsLoadErrors(results){
 const sections=[
  ['API kľúče','settings-api-key-panel','api-key-list'],
  ['API projekty',null,'settings-api-projects'],
  ['Trezor','settings-secret-panel','secret-list'],
  ['Modely','settings-ai-provider-panel','settings-ai-providers'],
  ['Peňaženka','settings-wallet-panel',null],
  ['Používatelia','settings-users-panel','settings-users'],
  ['Samozlepšovanie','settings-self-improvement-panel','self-improvement-list']
 ];
 const failed=[];
 results.forEach((result,i)=>{
  const [name,panelId,listId]=sections[i],panel=panelId?$(panelId):null,bad=result.status==='rejected'||i===3&&results[2].status==='rejected';
  if(panel){panel.inert=bad;panel.classList.toggle('settings-load-error',bad);}
  if(bad){failed.push(name+': '+(result.status==='rejected'?result.reason.message:'Trezor sa nenačítal'));if(listId)$(listId).replaceChildren(node('p','Táto sekcia sa nenačítala. Obnov nastavenia.','section-load-message'));}
 });
 const warning=$('settings-partial-errors');warning.hidden=!failed.length;warning.textContent=failed.join(' · ');
}
async function loadCapabilityReadiness(){
 const list=$('settings-readiness'),summary=$('settings-readiness-summary');
 summary.textContent='Načítavam potvrdený stav…';
 try{
  const {integrationPresentation}=await import('/settings-state.js'),data=await api('/capabilities');list.replaceChildren();
  const admin=['owner','admin'].includes(state?.account?.role);
  for(const item of data.groups||[]){
   const card=node('article',undefined,'readiness-card'),status=integrationPresentation(item.status);
   card.dataset.status=status.tone;card.append(node('h3',item.name),node('b',status.label),node('p',item.note));
   if(item.local&&!admin)card.append(node('small','Táto schopnosť vyžaduje správcovské oprávnenie.'));
   const action=node('button',item.local?'Ukázať stav PC brány':'Otvoriť chat');
   action.type='button';action.disabled=item.local&&!admin;
   action.onclick=safe(async()=>{if(item.local){$('settings-gateway').scrollIntoView({behavior:'smooth',block:'center'});}else await setView('chat');});
   card.append(action);list.append(card);
  }
  summary.textContent=(data.pc_connected?'PC brána pripojená':'PC brána nepripojená')+' · aktualizované '+new Date(data.checked_at).toLocaleTimeString('sk-SK');
 }catch(error){list.replaceChildren();summary.textContent='Pripravenosť sa nepodarilo načítať: '+error.message;}
}
