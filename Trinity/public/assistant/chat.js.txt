function formatMarkdown(text){
  if(!text)return '';
  const d=document.createElement('div');
  d.textContent=text;
  let s=d.innerHTML;
  s=s.replace(/```([a-zA-Z0-9_-]*)\n([\s\S]*?)```/g,'<pre class="code-block"><code class="lang-$1">$2</code></pre>');
  s=s.replace(/`([^`\n]+)`/g,'<code class="inline-code">$1</code>');
  s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
  s=s.replace(/(^|[^*])\*([^*\n]+)\*/g,'$1<em>$2</em>');
  s=s.replace(/^[ \t]*[\*\-][ \t]+(.+)$/gm,'<li>$1</li>');
  s=s.replace(/(<li>[\s\S]*?<\/li>)/g,'<ul class="md-list">$1</ul>');
  s=s.replace(/<\/ul>\s*<ul class="md-list">/g,'');
  s=s.replace(/\n\n+/g,'<br><br>');
  return s;
}
function renderReply(target,text){
  target.innerHTML=formatMarkdown(text);
  const match=text.match(/\/api\/assistant\/media\/images\/[0-9a-f-]{36}/i);
  if(match){
    const image=document.createElement('img');
    image.className='generated-media';
    image.src=match[0];
    image.alt='Obrázok vytvorený Trinity';
    image.loading='lazy';
    target.append(image);
  }
  const fileMatches=Array.from(text.matchAll(/\/api\/assistant\/files\/([0-9a-f-]{36})\/([a-zA-Z0-9._\-]+)/gi));
  if(fileMatches.length){
    for(const fm of fileMatches){
      const url=fm[0];
      const fname=fm[2];
      const ext=fname.split('.').pop()?.toLowerCase();
      let icon='📄';
      if(ext==='pdf')icon='📕';
      else if(ext==='py')icon='🐍';
      else if(ext==='json')icon='📦';
      else if(['js','ts','mjs'].includes(ext))icon='⚡';
      else if(['jpg','png','webp','gif','svg'].includes(ext))icon='🖼️';
      else if(['wav','mp3','ogg'].includes(ext))icon='🎵';
      else if(ext==='html'&&/video/i.test(fname))icon='🎬';
      else if(ext==='html'&&/prezentacia|slide/i.test(fname))icon='📊';
      else if(ext==='csv')icon='📊';
      const fileCard=document.createElement('div');
      fileCard.className='generated-file-card';
      fileCard.innerHTML=`<span class="file-icon">${icon}</span><div class="file-info"><b>${fname}</b><small>Vytvorený súbor v R2 archíve</small></div><a href="${url}?download=1" download="${fname}" class="file-download-btn">⬇ Stiahnuť</a><a href="${url}" target="_blank" rel="noopener" class="file-view-btn">Otvoriť ↗</a>`;
      target.append(fileCard);
      if(['wav','mp3','ogg'].includes(ext)){
        const audio=document.createElement('audio');
        audio.controls=true;
        audio.preload='metadata';
        audio.src=url;
        audio.style.cssText='width:100%;margin-top:8px;border-radius:10px';
        target.append(audio);
      }else if(['jpg','png','webp','gif','svg'].includes(ext)){
        const img=document.createElement('img');
        img.className='generated-media';
        img.src=url;
        img.alt=fname;
        img.loading='lazy';
        target.append(img);
      }else if(ext==='html'){
        const frame=document.createElement('iframe');
        frame.src=url;
        frame.title=fname;
        frame.setAttribute('sandbox','allow-scripts');
        frame.style.cssText='width:100%;aspect-ratio:16/9;min-height:320px;border:1px solid rgba(56,189,248,0.3);border-radius:12px;margin-top:10px;background:#050811';
        target.append(frame);
      }
    }
  }
}

function inspectStudioHtml(html, reviewData=null){
  const doc=new DOMParser().parseFromString(html,'text/html'),images=[...doc.images],inputs=[...doc.querySelectorAll('input,select,textarea')],checks=[['Úplný HTML dokument',/^\s*<!doctype html>/i.test(html)&&!!doc.querySelector('html')],['Názov stránky',!!doc.title.trim()],['Responzívny viewport',!!doc.querySelector('meta[name="viewport"]')],['Jasná nadpisová štruktúra',doc.querySelectorAll('h1').length===1],['Alternatívny text obrázkov',images.every(image=>image.hasAttribute('alt'))],['Popisy formulárov',inputs.every(input=>input.closest('label')||input.id&&doc.querySelector(`label[for="${CSS.escape(input.id)}"]`))],['Jazyk dokumentu',!!doc.documentElement.getAttribute('lang')],['Bez externých skriptov',[...doc.scripts].every(script=>!script.src)]];
  const passed=checks.filter(([,ok])=>ok).length,score=Math.round(passed/checks.length*100),list=$('studio-quality-list');
  list.replaceChildren();
  for(const [label,ok] of checks){const row=node('div',undefined,'quality-item '+(ok?'pass':'warn'));row.append(node('span',ok?'✓':'!'),node('b',label),node('small',ok?'Splnené':'Odporúčaná úprava'));list.append(row);}
  $('studio-quality-value').textContent=score+'%';
  $('studio-quality-score').textContent=score+'%';
  const rev = reviewData || currentStudio?.latest_review;
  const pScore = rev?.score_performance ?? Math.max(70, score - 5);
  const aScore = rev?.score_accessibility ?? score;
  const bScore = rev?.score_best_practices ?? Math.min(100, score + 5);
  const sScore = rev?.score_seo ?? (doc.title && doc.querySelector('meta[name="viewport"]') ? 100 : 75);
  if($('score-perf'))$('score-perf').textContent = pScore + '%';
  if($('score-a11y'))$('score-a11y').textContent = aScore + '%';
  if($('score-sec'))$('score-sec').textContent = bScore + '%';
  if($('score-seo'))$('score-seo').textContent = sScore + '%';
  const testBox = $('studio-generated-tests');
  if(testBox){
    if(rev?.generated_tests){
      testBox.value = rev.generated_tests;
    } else {
      const title = doc.title || 'Untitled';
      testBox.value = `import { test, expect } from '@playwright/test';\n\ntest.describe('AI Studio: ${title.replace(/'/g, "\\'")}', () => {\n  test('has correct title and heading', async ({ page }) => {\n    await page.goto('/');\n    await expect(page).toHaveTitle(/${title.slice(0, 15).replace(/'/g, "\\'")}/i);\n    const h1 = page.locator('h1');\n    await expect(h1).toBeVisible();\n  });\n  test('accessibility and responsive viewport pass', async ({ page }) => {\n    await page.setViewportSize({ width: 375, height: 667 });\n    await page.goto('/');\n    const main = page.locator('main, body');\n    await expect(main).toBeVisible();\n  });\n});`;
    }
  }
  const copyBtn = $('studio-copy-tests');
  if(copyBtn && !copyBtn._bound){
    copyBtn._bound = true;
    copyBtn.onclick = () => {
      if(testBox)navigator.clipboard.writeText(testBox.value);
      notice('Playwright testy boli skopírované do schránky.');
    };
  }
  return score;
}

async function loadTrinityDocuments(){
  const list=$('trinity-doc-list');if(!list)return;
  try{
    const q=new URLSearchParams();
    if(docCategoryFilter)q.set('category',docCategoryFilter);
    if(docSearchQuery)q.set('search',docSearchQuery);
    q.set('limit','60');
    const data=await api('/documents?'+q.toString());
    renderTrinityDocuments(data);
  }catch(e){
    list.replaceChildren(node('p','Načítavanie Trinity Dokumentov zlyhalo: '+e.message,'skeleton'));
  }
}

let currentChartSymbol = 'XAU-EUR';
let currentChartTimeframe = '1D';
let currentChartMode = 'area'; // 'area' or 'candles'
let cachedChartData = null;

async function loadCommodityChart(symbol = currentChartSymbol, timeframe = currentChartTimeframe) {
  currentChartSymbol = symbol;
  currentChartTimeframe = timeframe;
  try {
    const data = await api(`/commodities/chart?symbol=${encodeURIComponent(symbol)}&timeframe=${encodeURIComponent(timeframe)}`);
    cachedChartData = data;

    const titleEl = $('chart-active-title');
    if (titleEl) titleEl.textContent = `${data.symbol} · ${data.name}${data.available === false ? ' (Bez overeného zdroja)' : ''}`;

    const fmt = v => new Intl.NumberFormat('sk-SK', { style: 'currency', currency: 'EUR' }).format(Number(v || 0));

    if (data.available === false) {
      if ($('term-price')) $('term-price').textContent = '—';
      if ($('term-change')) {
        $('term-change').textContent = data.reason || 'Bez overeného zdroja';
        $('term-change').className = '';
      }
      renderOrderBook(null, 0);
      return;
    }

    if ($('term-price')) $('term-price').textContent = fmt(data.current_price);
    if ($('term-change')) {
      const chg = Number(data.change_24h_pct || 0);
      $('term-change').textContent = `${chg >= 0 ? '+' : ''}${chg.toFixed(2)}%`;
      $('term-change').className = chg >= 0 ? 'up' : 'down';
    }
    if ($('term-high')) $('term-high').textContent = fmt(data.high_24h);
    if ($('term-low')) $('term-low').textContent = fmt(data.low_24h);
    if ($('term-rsi')) $('term-rsi').textContent = String(data.technical_indicators?.rsi_14 ?? data.rsi_14 ?? '—');
    if ($('term-macd')) {
      const macd = (data.technical_indicators?.macd || data.macd_signal || 'NEUTRAL').toUpperCase();
      $('term-macd').textContent = macd;
      $('term-macd').className = macd.includes('BULL') ? 'up' : macd.includes('BEAR') ? 'down' : '';
    }
    if ($('term-spread')) $('term-spread').textContent = data.source ? 'Overené (Yahoo/ECB)' : 'Reálne burzové dáta';

    const buyBtn = $('term-buy-btn');
    if (buyBtn) buyBtn.textContent = `🟢 Kúpiť / Long (${(data.symbol || '').split('-')[0]})`;

    const tradeSym = $('commodity-trade-symbol');
    if (tradeSym) tradeSym.value = data.symbol;

    document.querySelectorAll('.chart-tf-btn[data-chart-tf]').forEach(b => {
      b.classList.toggle('active', b.dataset.chartTf === timeframe);
    });
    document.querySelectorAll('.chart-sym-pill[data-chart-symbol]').forEach(p => {
      p.classList.toggle('active', p.dataset.chartSymbol === symbol);
    });

    renderCommodityChart(data);
    renderOrderBook(data.order_book, data.current_price);
  } catch (e) {
    console.error('loadCommodityChart error:', e);
  }
}

function renderCommodityChart(data) {
  if (!data) return;
  const svg = $('comm-chart-svg');
  if (!svg) return;

  const width = 800;
  const height = 360;
  const padLeft = 24;
  const padRight = 72;
  const padTop = 20;
  const padBottom = 40;
  const plotW = width - padLeft - padRight;
  const plotH = height - padTop - padBottom;

  const candles = (data.candles || []).length ? data.candles : (data.sparkline || []).map((p, i) => ({
    time: new Date(Date.now() - (data.sparkline.length - i) * 3600000).toISOString(),
    open: p, high: p * 1.002, low: p * 0.998, close: p, vol: 50
  }));

  if (!candles.length) return;

  let minP = Math.min(...candles.map(c => c.low || c.close));
  let maxP = Math.max(...candles.map(c => c.high || c.close));
  if (minP === maxP) { minP *= 0.99; maxP *= 1.01; }
  const rangeP = maxP - minP;
  minP -= rangeP * 0.03;
  maxP += rangeP * 0.03;

  const getY = p => (padTop + plotH) - ((p - minP) / (maxP - minP)) * plotH;
  const getX = i => padLeft + (i / Math.max(1, candles.length - 1)) * plotW;

  const gridLayer = $('chart-grid-layer');
  if (gridLayer) {
    gridLayer.replaceChildren();
    for (let t = 0; t <= 4; t++) {
      const frac = t / 4;
      const priceTick = minP + frac * (maxP - minP);
      const y = getY(priceTick);

      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.setAttribute('x1', String(padLeft));
      line.setAttribute('y1', String(y));
      line.setAttribute('x2', String(width - padRight));
      line.setAttribute('y2', String(y));
      line.setAttribute('stroke', 'rgba(255,255,255,0.07)');
      line.setAttribute('stroke-dasharray', '3,3');
      gridLayer.append(line);

      const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      txt.setAttribute('x', String(width - padRight + 8));
      txt.setAttribute('y', String(y + 3.5));
      txt.setAttribute('fill', '#94a3b8');
      txt.setAttribute('font-size', '10');
      txt.setAttribute('font-family', 'ui-monospace,monospace');
      txt.textContent = priceTick >= 100 ? priceTick.toFixed(1) : priceTick.toFixed(2);
      gridLayer.append(txt);
    }
  }

  const volLayer = $('chart-vol-layer');
  if (volLayer) {
    volLayer.replaceChildren();
    const maxV = Math.max(...candles.map(c => c.vol || 1), 1);
    const colW = Math.max(2, (plotW / candles.length) * 0.6);
    candles.forEach((c, i) => {
      const x = getX(i);
      const vH = ((c.vol || 1) / maxV) * 38;
      const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
      rect.setAttribute('x', String(x - colW / 2));
      rect.setAttribute('y', String(padTop + plotH - vH));
      rect.setAttribute('width', String(colW));
      rect.setAttribute('height', String(vH));
      rect.setAttribute('fill', (c.close >= c.open) ? 'rgba(16,185,129,0.22)' : 'rgba(239,68,68,0.22)');
      rect.setAttribute('rx', '1');
      volLayer.append(rect);
    });
  }

  const areaPath = $('chart-area-path');
  const linePath = $('chart-line-path');
  const candleLayer = $('chart-candles-layer');

  if (currentChartMode === 'candles') {
    if (areaPath) areaPath.setAttribute('d', '');
    if (linePath) linePath.setAttribute('d', '');
    if (candleLayer) {
      candleLayer.replaceChildren();
      const candleW = Math.max(3, (plotW / candles.length) * 0.68);
      candles.forEach((c, i) => {
        const x = getX(i);
        const yHigh = getY(c.high);
        const yLow = getY(c.low);
        const yOpen = getY(c.open);
        const yClose = getY(c.close);
        const isUp = c.close >= c.open;
        const color = isUp ? '#10b981' : '#ef4444';

        const wick = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        wick.setAttribute('x1', String(x));
        wick.setAttribute('y1', String(yHigh));
        wick.setAttribute('x2', String(x));
        wick.setAttribute('y2', String(yLow));
        wick.setAttribute('stroke', color);
        wick.setAttribute('stroke-width', '1.2');
        candleLayer.append(wick);

        const bodyY = Math.min(yOpen, yClose);
        const bodyH = Math.max(2, Math.abs(yClose - yOpen));
        const body = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
        body.setAttribute('x', String(x - candleW / 2));
        body.setAttribute('y', String(bodyY));
        body.setAttribute('width', String(candleW));
        body.setAttribute('height', String(bodyH));
        body.setAttribute('fill', color);
        body.setAttribute('rx', '1.5');
        candleLayer.append(body);
      });
    }
  } else {
    if (candleLayer) candleLayer.replaceChildren();
    const pts = candles.map((c, i) => ({ x: getX(i), y: getY(c.close) }));
    if (pts.length) {
      const lineD = pts.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x.toFixed(1)} ${p.y.toFixed(1)}`).join(' ');
      const areaD = `${lineD} L ${pts[pts.length - 1].x.toFixed(1)} ${(padTop + plotH).toFixed(1)} L ${pts[0].x.toFixed(1)} ${(padTop + plotH).toFixed(1)} Z`;
      if (linePath) linePath.setAttribute('d', lineD);
      if (areaPath) areaPath.setAttribute('d', areaD);
    }
  }

  const labelLayer = $('chart-labels-layer');
  if (labelLayer) {
    labelLayer.replaceChildren();
    const step = Math.max(1, Math.floor(candles.length / 5));
    for (let i = 0; i < candles.length; i += step) {
      const c = candles[i];
      const d = new Date(c.time);
      const timeStr = currentChartTimeframe === '1D'
        ? d.toLocaleTimeString('sk-SK', { hour: '2-digit', minute: '2-digit' })
        : d.toLocaleDateString('sk-SK', { day: '2-digit', month: '2-digit' });

      const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      txt.setAttribute('x', String(getX(i)));
      txt.setAttribute('y', String(height - 12));
      txt.setAttribute('fill', '#64748b');
      txt.setAttribute('font-size', '10');
      txt.setAttribute('text-anchor', 'middle');
      txt.setAttribute('font-family', 'ui-monospace,monospace');
      txt.textContent = timeStr;
      labelLayer.append(txt);
    }
  }
}

function renderOrderBook(ob, currentPrice) {
  const asksEl = $('ob-asks-list');
  const bidsEl = $('ob-bids-list');
  if (!asksEl || !bidsEl) return;

  const fmt = v => new Intl.NumberFormat('sk-SK', { style: 'currency', currency: 'EUR' }).format(Number(v || 0));

  if ($('ob-mid-val')) $('ob-mid-val').textContent = fmt(currentPrice);
  if ($('ob-spread-badge')) $('ob-spread-badge').textContent = ob?.spread_pct ? `Spread: ${ob.spread_pct}%` : 'Reálne burzové dáta';

  const asks = (ob?.asks || []).slice(0, 5);
  const bids = (ob?.bids || []).slice(0, 5);

  if (!asks.length && !bids.length) {
    asksEl.replaceChildren();
    bidsEl.replaceChildren(node('div', 'Hĺbka trhu vyžaduje Level-2 burzové API; Trinity nezobrazuje vymyslené objednávky.', 'subtext'));
    return;
  }

  const maxTotal = Math.max(...asks.map(a => a.total || 0), ...bids.map(b => b.total || 0), 1);

  asksEl.replaceChildren();
  asks.forEach(a => {
    const row = node('div', undefined, 'ob-row');
    const bar = node('div', undefined, 'ob-depth-bar');
    bar.style.width = `${Math.min(100, Math.round((a.total / maxTotal) * 100))}%`;
    bar.style.background = 'rgba(239,68,68,0.18)';
    const pSpan = node('span', fmt(a.price));
    pSpan.style.color = '#ef4444';
    const aSpan = node('span', String(a.qty || a.amount || ''));
    row.append(bar, pSpan, aSpan);
    asksEl.append(row);
  });

  bidsEl.replaceChildren();
  bids.forEach(b => {
    const row = node('div', undefined, 'ob-row');
    const bar = node('div', undefined, 'ob-depth-bar');
    bar.style.width = `${Math.min(100, Math.round((b.total / maxTotal) * 100))}%`;
    bar.style.background = 'rgba(16,185,129,0.18)';
    const pSpan = node('span', fmt(b.price));
    pSpan.style.color = '#10b981';
    const aSpan = node('span', String(b.qty || b.amount || ''));
    row.append(bar, pSpan, aSpan);
    bidsEl.append(row);
  });
}

    openBtn.onclick = () => window.open(creation.data_uri || creation.url, '_blank');
    actRow.append(dl, openBtn);

    wrap.append(img, info, actRow);
  } else if (type === 'presentation') {
    const frame = document.createElement('iframe');
    frame.srcdoc = creation.html_deck || '';
    frame.style.width = '100%';

  } else {
    const content = node('div', undefined, 'trinity-doc-markdown');
    content.innerHTML = formatMarkdown(creation.content || creation.summary || JSON.stringify(creation, null, 2));
    wrap.append(content);
  }

  body.append(wrap);
  stage.hidden = false;
  stage.scrollIntoView({ behavior: 'smooth', block: 'start' });

async function loadMultimediaStudio() {
  const gallery = $('mm-gallery-list');
  if (!gallery) return;
  try {
    const res = await api('/multimedia', undefined, 'GET');
    const items = res.items || [];
    if ($('mm-total-badge')) $('mm-total-badge').textContent = `${items.length} DIEL V ARCHÍVE`;
    gallery.replaceChildren();

    if (!items.length) {
      gallery.append(node('p', 'Zatiaľ nemáš vygenerované žiadne multimédiá. Použi formuláre vyššie na vytvorenie videa, hudby, fotky alebo prezentácie.', 'skeleton'));
      return;
    }

    const typeIcons = { video: '🎬', music: '🎵', photo: '📸', presentation: '📊', creative_pack: '✨' };

    items.forEach(item => {
      const card = node('article', undefined, 'domain-market-card');
      const head = node('div', undefined, 'domain-market-head');
      head.append(
        node('strong', `${typeIcons[item.kind] || '✦'} ${item.title || item.topic}`),
        node('span', item.kind.toUpperCase(), 'domain-badge active')
      );
      const desc = node('p', item.prompt || item.topic || item.summary || '', 'domain-market-notes');
      const meta = node('small', `Vytvorené: ${new Date(item.created_at).toLocaleString('sk-SK')}`);
      const actions = node('div', undefined, 'domain-market-actions');

      const playBtn = node('button', '👁 Otvoriť a zobraziť', 'primary');
      playBtn.type = 'button';
      playBtn.onclick = () => displayMultimediaStage(item, item.kind);

      actions.append(playBtn);
      if (item.data_uri || item.audio_data_uri) {
        const dl = node('a', '⬇ Stiahnuť', 'button-link');
        dl.href = item.data_uri || item.audio_data_uri;
        dl.download = `${(item.title || 'trinity-media').replace(/[^a-z0-9_-]/gi, '_')}`;
        actions.append(dl);
      }
      card.append(head, desc, meta, actions);
      gallery.append(card);
    });
  } catch (e) {
    console.error('loadMultimediaStudio error:', e);
  }
}

let cachedWalletData = null;

function switchWalletTab(tabId) {
  document.querySelectorAll('#wallet-subnav [data-wallet-tab]').forEach(b => {
    b.classList.toggle('active', b.dataset.walletTab === tabId);
  });
  ['terminal', 'payout', 'config', 'history'].forEach(id => {
    const pane = $('wallet-pane-' + id);
    if (pane) pane.hidden = id !== tabId;
  });
}

async function loadWalletView() {
  try {
    const data = await api('/wallet/terminal/config', undefined, 'GET');
    cachedWalletData = data;
    renderWalletView(data);
  } catch (e) {
    console.error('loadWalletView error:', e);
  }
}

function renderWalletView(data) {
  if (!data) return;
  const fmt = v => new Intl.NumberFormat('sk-SK', { style: 'currency', currency: 'EUR' }).format(Number(v || 0));

  const w = data.wallet || {};
  const t = data.terminals || {};

  const balance = w.evidenced_balance_eur != null ? w.evidenced_balance_eur : (w.balance_eur || 0);
  const available = w.evidenced_balance_eur != null ? w.evidenced_balance_eur : (w.available_payout_eur || 0);
  const received = w.evidenced_received_eur != null ? w.evidenced_received_eur : (w.total_received_eur || 0);
  const withdrawn = w.evidenced_paid_out_eur != null ? w.evidenced_paid_out_eur : (w.total_withdrawn_eur || 0);

  if ($('wstat-balance')) $('wstat-balance').textContent = fmt(balance);
  if ($('wstat-available')) $('wstat-available').textContent = fmt(available);
  if ($('wstat-received')) $('wstat-received').textContent = fmt(received);
  if ($('wstat-withdrawn')) $('wstat-withdrawn').textContent = fmt(withdrawn);

  // Pre-fill config form
  if ($('wcfg-iban')) $('wcfg-iban').value = t.sepa?.iban || '';
  if ($('wcfg-bic')) $('wcfg-bic').value = t.sepa?.bic || '';
  if ($('wcfg-name')) $('wcfg-name').value = t.sepa?.account_name || '';
  if ($('wcfg-revolut')) $('wcfg-revolut').value = t.revolut?.handle || '';
  if ($('wcfg-paypal')) $('wcfg-paypal').value = t.paypal?.account || t.paypal?.email || '';
  if ($('wcfg-crypto')) $('wcfg-crypto').value = t.crypto?.address || '';
  if ($('wcfg-crypto-net')) $('wcfg-crypto-net').value = t.crypto?.network || 'polygon';

  // Pre-fill payout target based on default
  const payoutType = $('payout-dest-type')?.value || 'bank_iban';
  updatePayoutTargetDefault(payoutType, t);

  // Render history list
  const hist = $('wallet-history-list');
  if (hist) {
    hist.replaceChildren();
    const txs = data.recent_transactions || [];
    const payouts = data.recent_payouts || [];
    const charges = data.recent_charges || [];

    if (!txs.length && !payouts.length && !charges.length) {
      hist.append(node('p', 'Zatiaľ žiadne zaznamenané transakcie ani výbery v peňaženke.', 'skeleton'));
    } else {
      payouts.forEach(po => {
        const row = node('article', undefined, 'api-key-item');
        const info = node('div', undefined, 'commerce-offer-info');
        const top = node('div', undefined, 'commerce-offer-top');
        const isExecuted = po.status === 'reported_sent';
        top.append(
          node('b', isExecuted ? `Overený výber: ${fmt(po.amount_eur)} na ${po.destination_type.toUpperCase()}` : `Príprava výberu (NEODOSLANÉ): ${fmt(po.amount_eur)} na ${po.destination_type.toUpperCase()}`),
          node('span', isExecuted ? 'ODOSLANÉ (POTVRDENÉ)' : 'NEODOSLANÉ (DRAFT)', isExecuted ? 'domain-badge active' : 'domain-badge expired'),
          node('span', po.reference || 'DRAFT', 'detail-chip')
        );
        const delBtn = node('button', '🗑 Zmazať záznam', 'button-link');
        delBtn.style.color = '#ef4444';
        delBtn.style.marginLeft = 'auto';
        delBtn.onclick = safe(async () => {
          await api(`/wallet/payout/${po.id}/cancel`, {});
          notice(`Záznam výberu ${po.id} bol trvalo vymazaný.`);
          await loadWalletView();
        });
        top.append(delBtn);
        info.append(top, node('small', `Cieľ: ${po.target}${po.note ? ' · ' + po.note : ''} · Stav: ${isExecuted ? 'Odoslané a overené bankovým výpisom' : 'Iba pripravená inštrukcia (Trinity peniaze neodoslala)'} · ${new Date(po.created_at).toLocaleString('sk-SK')}`));
        row.append(info);
        hist.append(row);
      });

      charges.forEach(ch => {
        const row = node('article', undefined, 'api-key-item');
        const info = node('div', undefined, 'commerce-offer-info');
        const top = node('div', undefined, 'commerce-offer-top');
        top.append(
          node('b', `Príkaz na príjem: ${fmt(ch.amount_eur)} — ${ch.purpose}`),
          node('span', 'PLATOBNÝ PRÍKAZ', 'domain-badge active'),
          node('span', ch.variable_symbol ? `VS: ${ch.variable_symbol}` : 'TERMINÁL', 'detail-chip')
        );
        info.append(top, node('small', `Terminál: ${ch.terminal_type.toUpperCase()} · Stav: ${ch.status} · ${new Date(ch.created_at).toLocaleString('sk-SK')}`));
        row.append(info);
        hist.append(row);
      });

      txs.forEach(t => {
        const row = node('article', undefined, 'api-key-item');
        const info = node('div', undefined, 'commerce-offer-info');
        const top = node('div', undefined, 'commerce-offer-top');
        const isDep = t.type === 'deposit';
        top.append(
          node('b', `${isDep ? 'Vklad / Príjem' : 'Platba'}: ${fmt(t.amount_eur)}`),
          node('span', t.type.toUpperCase(), isDep ? 'domain-badge active' : 'domain-badge for_sale'),
          node('span', `Zostatok: ${fmt(t.balance_after_eur)}`, 'detail-chip')
        );
        info.append(top, node('small', `${t.recipient ? 'Príjemca: ' + t.recipient + ' · ' : ''}${t.note || ''} · ${new Date(t.created_at).toLocaleString('sk-SK')}`));
        row.append(info);
        hist.append(row);
      });
    }
  }
}

function updatePayoutTargetDefault(type, terms) {
  const targetInput = $('payout-target');
  if (!targetInput) return;
  if (type === 'bank_iban') {
    targetInput.placeholder = 'napr. SK890200000000...';
    targetInput.value = terms?.sepa?.iban || '';
  } else if (type === 'paypal') {
    targetInput.placeholder = 'napr. mojemail@paypal.com';
    targetInput.value = terms?.paypal?.email || '';
  } else if (type === 'revolut') {
    targetInput.placeholder = 'napr. @mojtag alebo telefón';
    targetInput.value = terms?.revolut?.handle ? terms.revolut.handle.replace(/.*revolut\.me\//, '@') : '';
  } else if (type === 'crypto') {
    targetInput.placeholder = '0x... (Polygon / EVM adresa)';
    targetInput.value = terms?.crypto?.address || '';
  }
}

$('payout-dest-type')?.addEventListener('change', e => {
  if (cachedWalletData) updatePayoutTargetDefault(e.target.value, cachedWalletData.terminals);
});

let cachedGraphData = null;
let selectedGraphNode = null;

async function loadGraphView() {
  try {
    const data = await api('/memory/graph');
    cachedGraphData = data;
    renderGraphStats(data);
    renderGraphSvg(data);
  } catch (err) {

function renderGraphStats(data) {
  const nodes = data.nodes || [];
  const edges = data.edges || [];
  if ($('graph-stat-nodes')) $('graph-stat-nodes').textContent = data.count_nodes ?? nodes.length;
  if ($('graph-stat-edges')) $('graph-stat-edges').textContent = data.count_edges ?? edges.length;
  if ($('graph-stat-domains')) $('graph-stat-domains').textContent = nodes.filter(n => n.type === 'domain').length;
  if ($('graph-stat-projects')) $('graph-stat-projects').textContent = nodes.filter(n => n.type === 'studio_project').length;
  if ($('graph-stat-txs')) $('graph-stat-txs').textContent = nodes.filter(n => n.type === 'transaction').length;
}

function renderGraphSvg(data) {
  const svg = $('graph-svg');
  if (!svg) return;
  svg.innerHTML = '';

  const filterType = $('graph-filter-type')?.value || 'all';
  const searchQuery = ($('graph-search')?.value || '').trim().toLowerCase();

  let nodes = data.nodes || [];
  if (filterType !== 'all') {
    nodes = nodes.filter(n => n.type === filterType || n.type === 'owner');
  }
  if (searchQuery) {
    nodes = nodes.filter(n => 
      n.label.toLowerCase().includes(searchQuery) ||
      n.type.toLowerCase().includes(searchQuery) ||
      JSON.stringify(n.properties || {}).toLowerCase().includes(searchQuery)
    );
  }

  const nodeMap = new Map();
  nodes.forEach(n => nodeMap.set(n.id, n));

  const edges = (data.edges || []).filter(e => nodeMap.has(e.source_id) && nodeMap.has(e.target_id));

  const width = svg.clientWidth || 800;
  const height = svg.clientHeight || 550;
  const cx = width / 2;
  const cy = height / 2;

  const colorMap = {
    owner: '#e11d48',
    domain: '#38bdf8',
    studio_project: '#a78bfa',
    memory_note: '#34d399',
    transaction: '#fbbf24',
    system: '#94a3b8'
  };

  const nonOwnerNodes = nodes.filter(n => n.type !== 'owner');
  const ownerNode = nodes.find(n => n.type === 'owner') || { id: 'owner', label: 'Ivan Sabo (Basterix)', type: 'owner' };

  ownerNode.x = cx;
  ownerNode.y = cy;
  ownerNode.color = colorMap.owner;

  const total = nonOwnerNodes.length;
  const radius = Math.min(width, height) * 0.38;

  nonOwnerNodes.forEach((n, idx) => {
    const angle = (idx / Math.max(1, total)) * 2 * Math.PI - Math.PI / 2;
    const r = radius * (0.8 + 0.35 * (idx % 3));
    n.x = cx + r * Math.cos(angle);
    n.y = cy + r * Math.sin(angle);
    n.color = colorMap[n.type] || '#60a5fa';
  });

  const allRenderNodes = [ownerNode, ...nonOwnerNodes];
  const allNodeMap = new Map(allRenderNodes.map(n => [n.id, n]));

  // Draw Edges
  edges.forEach(e => {
    const s = allNodeMap.get(e.source_id);
    const t = allNodeMap.get(e.target_id);
    if (!s || !t) return;
    const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    line.setAttribute('x1', s.x);
    line.setAttribute('y1', s.y);
    line.setAttribute('x2', t.x);
    line.setAttribute('y2', t.y);
    line.setAttribute('stroke', '#334155');
    line.setAttribute('stroke-width', '1.5');
    line.setAttribute('stroke-dasharray', '4 2');
    svg.appendChild(line);

    if (e.relationship && Math.hypot(t.x - s.x, t.y - s.y) > 80) {
      const midX = (s.x + t.x) / 2;
      const midY = (s.y + t.y) / 2;
      const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      text.setAttribute('x', midX);
      text.setAttribute('y', midY - 3);
      text.setAttribute('fill', '#64748b');
      text.setAttribute('font-size', '10');
      text.setAttribute('text-anchor', 'middle');
      text.textContent = e.relationship;
      svg.appendChild(text);
    }
  });

  // Draw Nodes
  allRenderNodes.forEach(n => {
    const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    g.style.cursor = 'pointer';

    const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle.setAttribute('cx', n.x);
    circle.setAttribute('cy', n.y);
    circle.setAttribute('r', n.type === 'owner' ? '22' : '15');
    circle.setAttribute('fill', n.color);
    circle.setAttribute('stroke', '#0f172a');
    circle.setAttribute('stroke-width', '2.5');

    const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
    text.setAttribute('x', n.x);
    text.setAttribute('y', n.y + (n.type === 'owner' ? 34 : 26));
    text.setAttribute('fill', '#e2e8f0');
    text.setAttribute('font-size', n.type === 'owner' ? '12' : '11');
    text.setAttribute('font-weight', n.type === 'owner' ? 'bold' : 'normal');
    text.setAttribute('text-anchor', 'middle');
    text.textContent = n.label.length > 20 ? n.label.slice(0, 18) + '…' : n.label;

    g.appendChild(circle);
    g.appendChild(text);

    g.onclick = () => selectGraphNode(n, edges, allNodeMap);
    svg.appendChild(g);
  });
}

function selectGraphNode(nodeData, allEdges, nodeMap) {
  selectedGraphNode = nodeData;
  const title = $('graph-node-title');
  const details = $('graph-node-details');
  const actions = $('graph-node-actions');
  if (title) title.textContent = nodeData.label;

  const connectedEdges = allEdges.filter(e => e.source_id === nodeData.id || e.target_id === nodeData.id);
  const connectionsHtml = connectedEdges.map(e => {
    const isSource = e.source_id === nodeData.id;
    const otherId = isSource ? e.target_id : e.source_id;
    const otherNode = nodeMap.get(otherId);
    const targetLabel = otherNode?.label || otherId;
    return `<li><b>${e.relationship}</b> ${isSource ? '→' : '←'} <span>${targetLabel}</span></li>`;
  }).join('');

  const props = nodeData.properties || {};
  const propsHtml = Object.entries(props).map(([k, v]) => `<div><span style="color:var(--text-muted,#94a3b8)">${k}:</span> <b>${typeof v === 'object' ? JSON.stringify(v) : v}</b></div>`).join('');

  if (details) {
    details.innerHTML = `
      <div style="margin-bottom:8px">
        <span class="detail-chip" style="background:rgba(255,255,255,0.08);margin-bottom:6px;display:inline-block">${nodeData.type.toUpperCase()}</span>
        <div style="font-size:12px;color:var(--text-muted,#94a3b8)">ID: ${nodeData.id}</div>
      </div>
      <div style="background:rgba(0,0,0,0.2);padding:8px;border-radius:6px;margin:8px 0;font-size:12px">
        ${propsHtml || '<em style="color:#64748b">Žiadne dodatočné vlastnosti</em>'}
      </div>
      <strong>Prepojenia (${connectedEdges.length}):</strong>
      <ul style="padding-left:18px;margin:6px 0;font-size:12px">${connectionsHtml || '<li>Žiadne priame hrany</li>'}</ul>
    `;
  }
  if (actions) actions.style.display = 'flex';
}

let liveVoiceStream = null;
let liveVoiceRecorder = null;
let liveVoiceChunks = [];
let liveVoiceActive = false;

$('voice-dialogue-btn')?.addEventListener('click', safe(async () => {
  const btn = $('voice-dialogue-btn');
  if (liveVoiceActive) {
    liveVoiceActive = false;
    btn.classList.remove('recording');
    btn.textContent = '⌛ Spracúvam reč…';
    if (liveVoiceRecorder && liveVoiceRecorder.state === 'recording') {
      liveVoiceRecorder.stop();
    }
    return;
  }

  if (!navigator.mediaDevices?.getUserMedia) {
    throw new Error('Váš prehliadač nepodporuje nahrávanie mikrofónu cez getUserMedia.');
  }

  liveVoiceStream = await navigator.mediaDevices.getUserMedia({ audio: true });
  liveVoiceChunks = [];
  liveVoiceActive = true;
  btn.classList.add('recording');
  btn.textContent = '🛑 Hovorím... (Klikni pre odoslanie)';

  liveVoiceRecorder = new MediaRecorder(liveVoiceStream);
  liveVoiceRecorder.ondataavailable = e => {
    if (e.data.size) liveVoiceChunks.push(e.data);
  };

  liveVoiceRecorder.onstop = async () => {
    liveVoiceStream.getTracks().forEach(t => t.stop());
    const audioBlob = new Blob(liveVoiceChunks, { type: liveVoiceRecorder.mimeType || 'audio/webm' });

    try {
      notice('Odosielam hlasový vstup pre Trinity Voice…');
      const buffer = await audioBlob.arrayBuffer();
      const bytes = new Uint8Array(buffer);
      let binary = '';
      for (let i = 0; i < bytes.length; i++) {
        binary += String.fromCharCode(bytes[i]);
      }
      const base64Audio = btoa(binary);

      const resp = await fetch('/api/assistant/voice/dialogue', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          audio_base64: base64Audio,
          audio_mime: liveVoiceRecorder.mimeType || 'audio/webm'
        })
      });

      const res = await resp.json();
      if (!res.success) throw new Error(res.error || 'Hlasový dialóg zlyhal.');

      if (res.transcription) {
        bubble('user', res.transcription);
      }
      const reply = bubble('assistant');
      renderReply(reply.content, res.text);
      bottom();

      if (res.audio_base64) {
        const audio = new Audio(`data:${res.audio_mime || 'audio/wav'};base64,${res.audio_base64}`);
        audio.play().catch(() => {});
      } else if ('speechSynthesis' in window) {
        const utter = new SpeechSynthesisUtterance(res.text);
        utter.lang = uiLanguage() === 'sk' ? 'sk-SK' : 'en-US';
        window.speechSynthesis.speak(utter);
      }

      notice('Trinity odpovedala hlasom.');
    } catch (err) {
      notice('Chyba hlasového dialógu: ' + err.message);
    } finally {
      btn.classList.remove('recording');
      btn.textContent = '🎙️ Live Hovor';
      liveVoiceActive = false;
    }
  };

  liveVoiceRecorder.start();
}));

      );
      const actions = node('div', undefined, 'domain-card-actions');
      const landingBtn = node('button', '🌐 Otvoriť Landing Page');
      landingBtn.type = 'button';
      landingBtn.onclick = () => window.open('/broker/' + encodeURIComponent(d.domain_name), '_blank');
      actions.append(landingBtn);

      const settleBtn = node('button', '✓ Vysporiadať Escrow & DNS', 'primary');
      settleBtn.type = 'button';
      settleBtn.onclick = safe(async () => {
        const buyer = prompt(`Zadaj e-mail overeného kupujúceho pre ${d.domain_name}:`, 'klient@firma.sk');
        if (!buyer) return;
        const targetDns = prompt(`Cieľová IP / server kupujúceho:`, '192.0.2.1');
        if (!targetDns) return;

        notice('Vytváram a vysporadúvam escrow transakciu s prepisom DNS…');
        const dealRes = await api('/domains/escrow/inquire', {
          domain_name: d.domain_name,
          deal_type: 'instant_buy',
          amount: d.sale_price || 490,
          currency: 'EUR',
          buyer_email: buyer,
          target_dns: targetDns
        });
        await api('/domains/escrow/settle', {
          deal_id: dealRes.deal_id,
          payment_reference: 'verified_bank_transfer',
          target_dns: targetDns
        });
        notice(`Escrow úspešne dokončený! Doména ${d.domain_name} prevedená, DNS prepísané na ${targetDns}.`);
        await loadDomains();
      });
      actions.append(settleBtn);

      list.append(card);
    }
  } catch (err) {
    list.replaceChildren(node('p', 'Chyba načítania escrow: ' + err.message, 'skeleton'));

function renderNetWorthChart(history) {
  const svg = $('nw-chart-svg');
  if (!svg || !history.length) return;
  svg.innerHTML = '';

  const width = svg.clientWidth || 700;
  const height = svg.clientHeight || 160;
  const padding = 28;

  const values = history.map(h => h.net_worth);
  const min = Math.min(...values) * 0.95;
  const max = Math.max(...values) * 1.05;

  const points = history.map((h, i) => {
    const x = padding + (i / Math.max(1, history.length - 1)) * (width - 2 * padding);
    const y = height - padding - ((h.net_worth - min) / Math.max(1, max - min)) * (height - 2 * padding);
    return { x, y, val: h.net_worth, label: h.date };
  });

  const pathD = points.map((p, i) => (i === 0 ? `M ${p.x} ${p.y}` : `L ${p.x} ${p.y}`)).join(' ');

  // Draw area
  const areaD = `${pathD} L ${points[points.length - 1].x} ${height - padding} L ${points[0].x} ${height - padding} Z`;
  const area = document.createElementNS('http://www.w3.org/2000/svg', 'path');
  area.setAttribute('d', areaD);
  area.setAttribute('fill', 'rgba(56, 189, 248, 0.12)');
  svg.appendChild(area);

  // Draw line
  const line = document.createElementNS('http://www.w3.org/2000/svg', 'path');
  line.setAttribute('d', pathD);
  line.setAttribute('fill', 'none');
  line.setAttribute('stroke', '#38bdf8');
  line.setAttribute('stroke-width', '2.5');
  svg.appendChild(line);

  // Draw dots and labels
  points.forEach(p => {
    const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle.setAttribute('cx', p.x);
    circle.setAttribute('cy', p.y);
    circle.setAttribute('r', '4');
    circle.setAttribute('fill', '#38bdf8');
    svg.appendChild(circle);

    const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
    txt.setAttribute('x', p.x);
    txt.setAttribute('y', height - 6);
    txt.setAttribute('fill', '#94a3b8');
    txt.setAttribute('font-size', '10');
    txt.setAttribute('text-anchor', 'middle');
    txt.textContent = p.label;
    svg.appendChild(txt);
  });
}

function bufferToBase64Url(buffer) {
  const bytes = new Uint8Array(buffer);
  let binary = '';
  for (let i = 0; i < bytes.byteLength; i++) binary += String.fromCharCode(bytes[i]);
  return btoa(binary).replace(/=/g, '').replace(/\+/g, '-').replace(/\//g, '_');
}

function base64UrlToBuffer(base64Url) {
  const binary = atob(String(base64Url || '').replace(/-/g, '+').replace(/_/g, '/'));
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
  return bytes.buffer;
}

async function initPasskeyLogin() {
  const btn = $('passkey-login-btn');
  const status = $('passkey-login-status');
  if (!btn) return;
  if (!window.PublicKeyCredential) {
    btn.style.display = 'none';
    return;
  }

  btn.onclick = async () => {
    try {
      if (status) status.textContent = 'Inicializujem biometrické overenie...';
      const optRes = await fetch('/api/auth/passkey/auth-options', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({})
      });
      if (!optRes.ok) throw new Error('Nepodarilo sa načítať autentifikačné možnosti.');
      const options = await optRes.json();
      options.challenge = base64UrlToBuffer(options.challenge);
      if (options.allowCredentials) {
        options.allowCredentials = options.allowCredentials.map(c => ({
          ...c,
          id: base64UrlToBuffer(c.id)
        }));
      }

      if (status) status.textContent = 'Prilož prst (TouchID) alebo použi Windows Hello...';
      const credential = await navigator.credentials.get({ publicKey: options });
      if (!credential) throw new Error('Biometrické overenie bolo zrušené.');

      if (status) status.textContent = 'Overujem biometrický podpis na serveri...';
      const verifyRes = await fetch('/api/auth/passkey/auth-verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          id: credential.id,
          rawId: bufferToBase64Url(credential.rawId),
          response: {
            clientDataJSON: bufferToBase64Url(credential.response.clientDataJSON),
            authenticatorData: bufferToBase64Url(credential.response.authenticatorData),
            signature: bufferToBase64Url(credential.response.signature),
            userHandle: credential.response.userHandle ? bufferToBase64Url(credential.response.userHandle) : null
          }
        })
      });

      const verifyData = await verifyRes.json();
      if (!verifyRes.ok || !verifyData.ok) {
        throw new Error(verifyData.error || 'Biometrické overenie zlyhalo.');
      }

      if (status) status.textContent = '✓ Úspešne prihlásený! Otváram Trinity...';
      location.reload();
    } catch (err) {
      if (status) status.textContent = '⚠️ ' + err.message;
    }
  };
}

async function loadPasskeys() {
  const container = $('passkey-list');
  if (!container) return;
  try {
    const res = await fetch('/api/auth/passkey/list');
    if (!res.ok) return;
    const data = await res.json();
    container.replaceChildren();
    if (!data.passkeys || !data.passkeys.length) {
      container.append(node('p', 'Zatiaľ nemáš zaregistrovaný žiadny Passkey na tomto účte.', 'skeleton'));
      return;
    }
    for (const pk of data.passkeys) {
      const row = node('div', undefined, 'api-key-item');
      const info = node('div');
      info.append(
        node('b', pk.friendly_name || 'Passkey Device'),
        node('small', `Vytvorené: ${pk.created_at ? new Date(pk.created_at).toLocaleDateString('sk-SK') : 'dnes'} · Použité: ${pk.counter}x · ID: ${pk.credential_id}`)
      );
      const delBtn = node('button', 'Odstrániť', 'danger');
      delBtn.onclick = async () => {
        if (!confirm(`Naozaj chceš zmazať Passkey "${pk.friendly_name}"?`)) return;
        await fetch(`/api/auth/passkey/${pk.id}`, { method: 'DELETE' });
        await loadPasskeys();
      };
      row.append(info, delBtn);
      container.append(row);
    }
  } catch (e) {
    console.error('loadPasskeys failed', e);
  }
}

async function initPasskeyRegistration() {
  const regBtn = $('register-passkey-btn');
  const status = $('passkey-reg-status');
  const nameInput = $('new-passkey-name');
  if (!regBtn) return;
  if (!window.PublicKeyCredential) {
    regBtn.disabled = true;
    if (status) status.textContent = 'Tento prehliadač nepodporuje WebAuthn.';
    return;
  }

  regBtn.onclick = async () => {
    try {
      if (status) status.textContent = 'Pripravujem registráciu kľúča...';
      const friendlyName = nameInput?.value.trim() || 'Moje zariadenie';
      const optRes = await fetch('/api/auth/passkey/register-options', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ friendlyName })
      });
      if (!optRes.ok) throw new Error('Nepodarilo sa vygenerovať registračnú výzvu.');
      const options = await optRes.json();
      options.challenge = base64UrlToBuffer(options.challenge);
      options.user.id = base64UrlToBuffer(options.user.id);
      if (options.excludeCredentials) {
        options.excludeCredentials = options.excludeCredentials.map(c => ({
          ...c,
          id: base64UrlToBuffer(c.id)
        }));
      }

      if (status) status.textContent = 'Potvrď registráciu na zariadení (TouchID / Windows Hello / YubiKey)...';
      const credential = await navigator.credentials.create({ publicKey: options });
      if (!credential) throw new Error('Registrácia bola zrušená.');

      if (status) status.textContent = 'Ukladám verejný kľúč do Trinity...';
      const verifyRes = await fetch('/api/auth/passkey/register-verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          id: credential.id,
          rawId: bufferToBase64Url(credential.rawId),
          response: {
            clientDataJSON: bufferToBase64Url(credential.response.clientDataJSON),
            attestationObject: bufferToBase64Url(credential.response.attestationObject),
            transports: credential.response.getTransports ? credential.response.getTransports() : ['internal', 'hybrid']
          },
          friendlyName
        })
      });

      const resData = await verifyRes.json();
      if (!verifyRes.ok || !resData.ok) {
        throw new Error(resData.error || 'Uloženie Passkey zlyhalo.');
      }

      if (status) status.textContent = '✓ Passkey úspešne zaregistrovaný!';
      if (nameInput) nameInput.value = '';
      await loadPasskeys();
    } catch (err) {
      if (status) status.textContent = '⚠️ ' + err.message;
    }
  };
}

async function loadAuditEvents() {
  const container = $('audit-events-list');
  if (!container) return;
  const search = $('audit-search-input')?.value.trim() || '';
  const category = $('audit-category-select')?.value || '';
  const severity = $('audit-severity-select')?.value || '';

  const params = new URLSearchParams();
  if (search) params.set('search', search);
  if (category) params.set('category', category);
  if (severity) params.set('severity', severity);
  params.set('limit', '50');

  const exportJsonBtn = $('audit-export-json-btn');
  const exportCsvBtn = $('audit-export-csv-btn');
  if (exportJsonBtn) exportJsonBtn.href = `/api/assistant/audit/export?format=json&${params.toString()}`;
  if (exportCsvBtn) exportCsvBtn.href = `/api/assistant/audit/export?format=csv&${params.toString()}`;

  try {
    const res = await fetch(`/api/assistant/audit/events?${params.toString()}`);
    if (!res.ok) throw new Error('Nepodarilo sa načítať auditný log.');
    const events = await res.json();
    container.replaceChildren();

    if (!events.length) {
      container.append(node('p', 'Žiadne auditné udalosti nezodpovedajú zvolenému filtru.', 'skeleton'));
      return;
    }

    for (const ev of events) {
      const card = node('article', undefined, 'incident-item');
      card.style.borderLeft = `4px solid ${ev.severity === 'critical' ? '#ef4444' : ev.severity === 'warning' ? '#f59e0b' : '#3b82f6'}`;
      card.style.marginBottom = '8px';
      card.style.padding = '10px 12px';

      const head = node('div', undefined, 'incident-head');
      const badgeCat = node('span', ev.category.toUpperCase(), 'detail-chip');
      const badgeSev = node('span', ev.severity.toUpperCase(), `detail-chip ${ev.severity}`);
      badgeSev.style.color = ev.severity === 'critical' ? '#ef4444' : ev.severity === 'warning' ? '#f59e0b' : '#3b82f6';

      head.append(node('b', ev.summary), node('div', undefined, 'chips', [badgeCat, badgeSev]));

      const meta = node('div', undefined, 'incident-meta');
      meta.style.fontSize = '12px';
      meta.style.color = 'var(--text-muted, #888)';
      meta.style.marginTop = '4px';
      meta.textContent = `Čas: ${new Date(ev.timestamp).toLocaleString('sk-SK')} · Aktér: ${ev.actor} · Akcia: ${ev.action}`;

      const detailsWrap = node('details');
      detailsWrap.style.marginTop = '6px';
      const detailsSummary = node('summary', 'Podrobnosti JSON');
      detailsSummary.style.cursor = 'pointer';
      detailsSummary.style.fontSize = '11px';
      const pre = node('pre');
      pre.style.fontSize = '11px';
      pre.style.overflowX = 'auto';
      pre.style.background = 'rgba(0,0,0,0.3)';
      pre.style.padding = '8px';
      pre.style.borderRadius = '4px';
      pre.textContent = JSON.stringify(ev.details, null, 2);
      detailsWrap.append(detailsSummary, pre);

      card.append(head, meta, detailsWrap);
      container.append(card);
    }
  } catch (err) {
    container.replaceChildren(node('p', '⚠️ ' + err.message, 'skeleton'));
  }
}
